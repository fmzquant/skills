//[trans]该示范主要用海龟交易法则，演示“头寸计算，最大仓位控制等资金管理”的编写方法|[/trans]
//[trans]编写示范中，只对示范重点内容语句进行了注释，其他语句请自行翻译，或者咨询客服|[/trans]


TR:=MAX(MAX((HIGH-LOW),ABS(REF(CLOSE,1)-HIGH)),ABS(REF(CLOSE,1)-LOW));//[trans]真实波幅|[/trans]
ATR:=MA(TR,26); //[trans]求26个周期内真实波幅的简单移动平均|[/trans]
TC..INTPART((MONEYTOT*0.01/(UNIT*ATR)));//[trans]根据权益的1%计算下单手数|[/trans]
MTC..4*TC; //[trans]总的持仓头寸|[/trans]
HH:=HV(H,20);
LL:=LV(L,20);
CROSSUP(C,HH)&&ISLASTBK=0&&ISLASTSK=0&&BARPOS>=26,BK(TC);//[trans]最新价超过20周期的最高值，首次买入开仓，手数为TC手|[/trans]
CROSSDOWN(C,LL)&&ISLASTBK=0&&ISLASTSK=0&&BARPOS>=26,SK(TC); //[trans]最新价跌破20周期的最低值，首次卖出开仓，手数为TC手|[/trans]
C>=BKPRICE+0.5*ATR&&BKVOL<MTC&&ISLASTBK,BK(TC);//[trans]价格在上次开仓的基础上上涨0.5倍ATR，在手数不超过4倍TC的时候，买入加仓TC手|[/trans]
C<=SKPRICE-0.5*ATR&&SKVOL<MTC&&ISLASTSK,SK(TC);//[trans]价格在上次开仓的基础上下跌0.5倍ATR，在手数不超过4倍TC的时候，卖出加仓TC手|[/trans]
C<=(BKPRICE-2*ATR)&&BKVOL>0,SP(BKVOL);//[trans]最新价小于开仓价减去2倍的ATR，止损平仓|[/trans]
C>=(SKPRICE+2*ATR)&&SKVOL>0,BP(SKVOL); //[trans]最新价大于开仓价加上2倍的ATR，止损平仓|[/trans]
CROSSUP(H,HV(H,10))&&SKVOL>0,BP(SKVOL);//[trans]最高价上穿10周期最高价，平仓|[/trans]
CROSSDOWN(L,LV(L,10))&&BKVOL>0,SP(BKVOL); //[trans]最低价下穿10周期的最低价，平仓|[/trans]
TRADE_AGAIN(10);
