//[trans]止损点差为SL,止赢点差为TP|[/trans]
A:=MINPRICE1;//[trans]取模组交易合约的最小变动价位|[/trans]
(C<=BKPRICE-SL*A||C>=BKPRICE+TP*A)&&BKPRICE>0,SP;//[trans]低于买开仓价10个点差，多头止损；高于买开仓价20个点差，多头止赢|[/trans]
(C>=SKPRICE+SL*A||C<=SKPRICE-TP*A)&&SKPRICE>0,BP;//[trans]高于卖开仓价10个点差，空头止损；低于卖开仓价20个点差，空头止赢|[/trans]
AUTOFILTER;
